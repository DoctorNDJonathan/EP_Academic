First cell: load the data (already in the notebook)

import glob
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# The diamonds dataset attached to this notebook (Kaggle dataset: shivam2503/diamonds)
path = glob.glob('/kaggle/input/**/diamonds.csv', recursive=True)[0]
df = pd.read_csv(path, index_col=0)   # the first column is only a row number

# Same category order as seaborn's built-in copy: best grade first
df['cut'] = pd.Categorical(df['cut'], ['Ideal', 'Premium', 'Very Good', 'Good', 'Fair'])
df['color'] = pd.Categorical(df['color'], ['D', 'E', 'F', 'G', 'H', 'I', 'J'])
df['clarity'] = pd.Categorical(df['clarity'], ['IF', 'VVS1', 'VVS2', 'VS1', 'VS2', 'SI1', 'SI2', 'I1'])
